Есть несколько способов ограничить 3D модель в рамках родительского контейнера. Вот основные подходы:

## 1. Настройка камеры и контролов

```javascript
import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls';

const container = document.getElementById('canvas-container');
const width = container.clientWidth;
const height = container.clientHeight;

// Создаем сцену, камеру и рендерер
const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(75, width / height, 0.1, 1000);
const renderer = new THREE.WebGLRenderer({ antialias: true });

renderer.setSize(width, height);
container.appendChild(renderer.domElement);

// Настройка OrbitControls с ограничениями
const controls = new OrbitControls(camera, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.05;

// Ограничиваем zoom
controls.minDistance = 2;
controls.maxDistance = 10;

// Ограничиваем углы вращения
controls.minPolarAngle = 0; // радианы
controls.maxPolarAngle = Math.PI; // радианы
```

## 2. Обработка изменения размера окна

```javascript
function onWindowResize() {
    const container = document.getElementById('canvas-container');
    const width = container.clientWidth;
    const height = container.clientHeight;
    
    camera.aspect = width / height;
    camera.updateProjectionMatrix();
    renderer.setSize(width, height);
}

window.addEventListener('resize', onWindowResize);
```

## 4. Полный пример с ограничением модели

```javascript
class BoundedScene {
    constructor(containerId) {
        this.container = document.getElementById(containerId);
        this.init();
        this.setupEventListeners();
    }

    init() {
        const width = this.container.clientWidth;
        const height = this.container.clientHeight;

        // Сцена
        this.scene = new THREE.Scene();
        this.scene.background = new THREE.Color(0xf0f0f0);

        // Камера
        this.camera = new THREE.PerspectiveCamera(75, width / height, 0.1, 1000);
        this.camera.position.z = 5;

        // Рендерер
        this.renderer = new THREE.WebGLRenderer({ antialias: true });
        this.renderer.setSize(width, height);
        this.container.appendChild(this.renderer.domElement);

        // Контролы
        this.controls = new OrbitControls(this.camera, this.renderer.domElement);
        this.setupControls();

        // Освещение
        this.addLights();

        // Модель
        this.addModel();

        // Анимация
        this.animate();
    }

    setupControls() {
        this.controls.enableDamping = true;
        this.controls.dampingFactor = 0.05;
        this.controls.minDistance = 2;
        this.controls.maxDistance = 8;
        this.controls.maxPolarAngle = Math.PI; // Запрещаем переворот модели
    }

    addLights() {
        const ambientLight = new THREE.AmbientLight(0x404040);
        this.scene.add(ambientLight);

        const directionalLight = new THREE.DirectionalLight(0xffffff, 0.5);
        directionalLight.position.set(1, 1, 1);
        this.scene.add(directionalLight);
    }

    addModel() {
        // Пример: простая геометрия
        const geometry = new THREE.BoxGeometry(1, 1, 1);
        const material = new THREE.MeshPhongMaterial({ color: 0x00ff00 });
        this.model = new THREE.Mesh(geometry, material);
        this.scene.add(this.model);
    }

    setupEventListeners() {
        window.addEventListener('resize', () => this.onResize());
    }

    onResize() {
        const width = this.container.clientWidth;
        const height = this.container.clientHeight;

        this.camera.aspect = width / height;
        this.camera.updateProjectionMatrix();
        this.renderer.setSize(width, height);
    }

    animate() {
        requestAnimationFrame(() => this.animate());
        this.controls.update();
        this.renderer.render(this.scene, this.camera);
    }
}

// Использование
const scene = new BoundedScene('canvas-container');
```

## 5. Дополнительные ограничения для сложных сцен

```javascript
// Для предотвращения выхода за границы при анимации
function keepInBounds(object, bounds) {
    const position = object.position;
    
    position.x = THREE.MathUtils.clamp(position.x, -bounds.x, bounds.x);
    position.y = THREE.MathUtils.clamp(position.y, -bounds.y, bounds.y);
    position.z = THREE.MathUtils.clamp(position.z, -bounds.z, bounds.z);
}

// В анимационном цикле
function animate() {
    requestAnimationFrame(animate);
    
    // Ограничиваем позицию модели
    keepInBounds(model, new THREE.Vector3(3, 3, 3));
    
    controls.update();
    renderer.render(scene, camera);
}
```

Эти методы обеспечат, чтобы ваша 3D модель оставалась в пределах родительского контейнера и была адаптивной к изменению размеров окна.
